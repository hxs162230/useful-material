 public List<int[]> getOptimizedUtilize(int maxTravelDist, List<int[]> forwardRoutes, List<int[]> returnRoutes) {
        // Sort both lists in a descending order by route distance
        Collections.sort(forwardRoutes, (a,b) -> (b[1] - a[1]));
        Collections.sort(returnRoutes, (a,b) -> (b[1] - a[1]));
                         
        // PQ to save all possible routes as int[3]{f.id, r.id, dist} in a descending order
        Queue<int[]> possibleRoutes = new PriorityQueue<int[]>(
            Math.max(forwardRoutes.size(), returnRoutes.size()),
            (a,b) -> (b[2] - a[2]));
                
        for(int[] forwardRoute: forwardRoutes) {
            if (forwardRoute[1] < maxTravelDist) {
                int retIndex = find(returnRoutes, maxTravelDist - forwardRoute[1]);
                if (retIndex > 0) {
                    int dist = forwardRoute[1] + returnRoutes.get(retIndex)[1];
                    possibleRoutes.offer(new int[]{forwardRoute[0], returnRoutes.get(retIndex)[0], dist});
                }
            }
        }
                         
        List<int[]> bestRoute = new ArrayList<int[]>();
        if (!possibleRoutes.isEmpty()) {
            int max = possibleRoutes.peek()[2];
            while(!possibleRoutes.isEmpty() && possibleRoutes.peek()[2] == max) {
                int[] route = possibleRoutes.poll();
                bestRoute.add(new int[]{route[0], route[1]});
            }
        }
    }

    private int find(List<int[]> routes, int v) {
        int s = 0;
        int e = routes.size()-1;
        int last = -1;
        while( s <= e ) {
            int mid = (s+e) / 2;
            int[] midRoute = routes.get(mid);
            if (midRoute[1] == v) {
                return mid;
            }
            if (v < midRoute[1]) {
                s = mid + 1;
            } else {
                // the last index where route is smaller than v
                last = mid;
                e = e - 1;
            }                       
        }
        return last;
    }

    private static int[] numberOfItems(String str, int[] starts, int[] ends) {
        int len = str.length();
        int[] result = new int[starts.length], left = new int[len], right = new int[len];
//        Left array will have the index of close\open located on the left side.
//        Right array will have the index of close\open located on the right side.
        int[] presum = new int[len];
        presum[0]= str.charAt(0)=='|'?0:1;
        int closeIdx = Integer.MAX_VALUE;
        for (int i = len - 1; i >= 0; i--) {
            if (str.charAt(i) == '|') closeIdx = i;
            right[i] = closeIdx;
        }
        closeIdx = Integer.MAX_VALUE;
        for (int i = 0; i < len; i++) {
            if (str.charAt(i) == '|') closeIdx = i;
            left[i] = closeIdx;
            if(i>=1){
                presum[i] = str.charAt(i)=='|' ? presum[i-1]:presum[i-1]+1;
            }
        }
        for (int i = 0; i < starts.length; i++) {
            int start = starts[i], end = ends[i];
            int startIdx = right[start - 1], endIdx = left[end - 1];
            result[i] = getCount(startIdx, endIdx, str, presum);
        }
        return result;
    }

    private static int getCount(int startIdx, int endIdx, String str,int[] presum) {
        if (startIdx == Integer.MAX_VALUE || endIdx == Integer.MAX_VALUE || startIdx >= endIdx) return 0;
        return presum[endIdx] - presum[startIdx];
    }
}

                pre = 1;
            } else {
                pre = 1;
            }
            time++;
        }

    public static int fiveStarReviews(int[][] productRatings, int ratingsThreshold) {
        int len = productRatings.length;
        PriorityQueue<Node> pq = new PriorityQueue<>((a,b)->Double.compare(b.val,a.val));
        double curval = 0;
        for(int[] p:productRatings){
            pq.add(new Node(p[0],p[1]));
        
            curval += (double)p[0]/p[1];
        }
        curval/=len;
        int res = 0;
        while(curval*100<ratingsThreshold){
            Node cur = pq.poll();
            double r = cur.overall(len);
            curval+=r;
            res++;
            pq.add(new Node(cur.x+1,cur.y+1));
        }
        
        return res;
    }

    }

                if (count + 1 > max) {
                    result = word;

                }

            }

            result = word;
        }

{  
    if (root == null)  
        root = newNode(key);  
    else if (root.key > key)  
        root.left = insert(root.left, key);  
    else if (root.key < key)  
        root.right = insert(root.right, key);  
    return root;  
}

// root. This function assumes that x exists  
// in BST and BST is not NULL.  
static int distanceFromRoot(Node root, int x)  
{  
    if (root.key == x)  
        return 0;  
    else if (root.key > x)  
        return 1 + distanceFromRoot(root.left, x);  
    return 1 + distanceFromRoot(root.right, x);  
}

// This function assumes that a and b exist  
// in BST.  
static int distanceBetween2(Node root, int a, int b)  
{  
    if (root == null)  
        return 0;  
  
    // Both keys lie in left  
    if (root.key > a && root.key > b)  
        return distanceBetween2(root.left, a, b);  
  
    // Both keys lie in right  
    if (root.key < a && root.key < b) // same path

    // Lie in opposite directions (Root is  
    // LCA of two nodes)  
    if (root.key >= a && root.key <= b)  
        return distanceFromRoot(root, a) + distanceFromRoot(root, b); 
          
    return 0; 
}

    private static int getMinScore(int nodes, int edges, int[] from, int[] to) {
        min = Integer.MAX_VALUE;
        Map<Integer, Set<Integer>> graph = new HashMap<>();
        for (int i = 0; i < edges; i++) {
            graph.computeIfAbsent(from[i], val -> new HashSet<>()).add(to[i]);
            graph.computeIfAbsent(to[i], val -> new HashSet<>()).add(from[i]);
        }
        boolean[] visited = new boolean[nodes + 1];
        for (int i = 1; i <= nodes; i++) {
            if (graph.containsKey(i)) {
                dfs(i, i, graph, visited, 0, 1);
            }
        }
        return min == Integer.MAX_VALUE ? -1 : min;
    }

    private static void dfs(int start, int cur, Map<Integer, Set<Integer>> graph, boolean[] visited, int score, int edge) {
        visited[cur] = true;
        Set<Integer> connections = graph.getOrDefault(cur, new HashSet<>());
        int curScore = connections.size() - edge, totalScore = score + curScore;
        for (int next : connections) {
            if (edge == 3) {
                if (next == start) {
                    min = Math.min(min, totalScore);
                }
            } else if (!visited[next] && edge < 3) {
                dfs(start, next, graph, visited, totalScore, edge + 1);
            }
        }
        visited[cur] = false;
    }

  public int countTeams(int num, int[] skills, int minAssociates, int minLevel, int maxLevel) {
    // 1. find associates with right level of skill
    int satisfied = 0;
    for (int skill : skills) {
      if (skill <= maxLevel && skill >= minLevel) {
        satisfied++;
      }
    }

    // 2. not enough associates to form the team
    if (satisfied < minAssociates) {
      return 0;
    }

    /*
    3. assemble the team
      3.1. choose minAssociates from satisfied
      3.2. choose minAssociates + 1 from satisfied
      ...
      3.k choose minAssociates + k from satisfied, where minAssociates + k == satisfied
    
    use combination algorithm. choose k from n,  C(n, k) = n!/ (n - k)! * k!
     */
    int[] factorial = new int[satisfied + 1];
    factorial[0] = 1;
    for (int i = 1; i <= satisfied; i++) {
      factorial[i] = factorial[i - 1] * i;
    }

    int total = 0;
    for (int i = minAssociates; i <= satisfied; i++) {
      total += factorial[satisfied] / factorial[i] * factorial[satisfied - i];
    }

    return total;
  }
